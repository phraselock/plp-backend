import io.javalin.Javalin;
import org.eclipse.jetty.server.Server;
import org.eclipse.jetty.server.HttpConfiguration;
import org.eclipse.jetty.server.ForwardedRequestCustomizer;

Javalin app = Javalin.create(config -> {
    config.jetty.modifyServer(server -> {
        HttpConfiguration httpConfig = new HttpConfiguration();
        httpConfig.addCustomizer(new ForwardedRequestCustomizer());

        server.getConnectors()[0].getConnectionFactory(HttpConfiguration.ConnectionFactory.class)
              .getHttpConfiguration()
              .addCustomizer(new ForwardedRequestCustomizer());
    });
});


import io.javalin.Javalin;
import org.eclipse.jetty.server.Server;
import org.eclipse.jetty.server.HttpConfiguration;
import org.eclipse.jetty.server.ForwardedRequestCustomizer;
import org.eclipse.jetty.server.HttpConnectionFactory;

Javalin app = Javalin.create(config -> {
    config.jetty.modifyServer(server -> {
        for (var connector : server.getConnectors()) {
            var connectionFactories = connector.getConnectionFactories();
            for (var cf : connectionFactories) {
                if (cf instanceof HttpConnectionFactory http) {
                    HttpConfiguration httpConfig = http.getHttpConfiguration();
                    httpConfig.addCustomizer(new ForwardedRequestCustomizer());
                }
            }
        }
    });
});
